find_package(OpenCASCADE 7.6.0 EXACT CONFIG REQUIRED)
get_target_property(GMESH_OCCT_LIBRARY_TYPE TKSTEP TYPE)
if(NOT GMESH_OCCT_LIBRARY_TYPE STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "STEP-enabled gmesh requires a static OCCT 7.6.0 installation")
endif()
add_library(gmesh_step STATIC app/main.cpp app/cli_options.cpp core/importer_registry.cpp core/boundary_triangulation.cpp formats/step/step_importer.cpp protocol/frame_writer.cpp)
target_link_libraries(gmesh_step PRIVATE CGAL::CGAL)
target_include_directories(gmesh_step PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(gmesh_step SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_compile_features(gmesh_step PUBLIC cxx_std_17)
target_compile_definitions(gmesh_step PRIVATE MODEL_IMPORT_ENABLE_STEP=1)
target_link_libraries(gmesh_step PRIVATE TKSTEP TKSTEP209 TKSTEPAttr TKSTEPBase TKXSBase TKShHealing TKMesh TKPrim TKTopAlgo TKGeomAlgo TKBRep TKGeomBase TKG3d TKG2d TKMath TKernel)
if(MSVC)
  target_compile_options(gmesh_step PRIVATE /utf-8)
endif()
if(GMESH_BUILD_TESTS)
  add_executable(gmesh_step_boundary_tests tests/test_boundary.cpp)
  target_link_libraries(gmesh_step_boundary_tests PRIVATE gmesh_step)
  add_executable(gmesh_step_protocol_tests tests/test_protocol.cpp protocol/frame_writer.cpp)
  target_include_directories(gmesh_step_protocol_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
endif()

if(GMESH_BUILD_TESTS)
  target_compile_features(gmesh_step_protocol_tests PRIVATE cxx_std_17)
endif()
